## 1. Configure a storage adapter

```ts
// dyrected.config.ts  (same for both frameworks)
import { LocalStorageAdapter } from '@dyrected/storage-local'

export default defineConfig({
  storage: new LocalStorageAdapter({ 
    uploadDir: './public/uploads',
    staticUrlPrefix: '/uploads'
  }),
  // For production use S3 — see Configuring Storage guide
})
```

See [Configuring Storage](/docs/guides/configuring-storage) for S3, R2, and Cloudinary setup.

---

## 2. Define an upload collection

```ts
// dyrected.config.ts  (same for both frameworks)
{
  slug: 'media',
  upload: {
    mimeTypes:   ['image/jpeg', 'image/png', 'image/webp'],
    maxFileSize: 5 * 1024 * 1024,
    imageSizes: [
      { name: 'thumbnail', width: 300, height: 300, fit: 'cover' },
      { name: 'card',      width: 800 },
    ],
  },
  fields: [
    { name: 'alt', type: 'text' },
  ],
}
```

## 3. Upload via the Admin UI

The Admin UI renders a full-featured media manager and file picker for upload collections. It supports three convenient upload methods:
1. **Browse**: Click the upload container to pick files via the standard browser selector.
2. **Drag & Drop**: Drag files from your computer and drop them anywhere on the Media page or Media Picker field preview to open the uploader.
3. **Clipboard Paste**: Copy any file or screenshot to your clipboard and press `Cmd+V` / `Ctrl+V` on the page to instantly load it into the uploader.

### Client-side Image Cropping

Both the Media Library dashboard and form-level `MediaPicker` fields support client-side image cropping:
- Hover over any image thumbnail in the forms to reveal a ✂️ **Scissors/Crop** button (shown for raster images like PNG, JPG, and WebP, but not SVGs or videos).
- Click the crop button on an image (or in the attachment details pane on the main media page) to open the crop dialog.
- Pick an aspect ratio preset (Free-form, 1:1, 4:3, 3:2, 16:9) and adjust the cropping handles.
- Click **Save** to apply the crop. The cropped version is uploaded as a new media document and the selection automatically updates to point to the cropped version. All image sizes are generated server-side automatically after upload.

## 4. Upload programmatically

<Tabs items={['SDK (any)', 'Next.js', 'Nuxt.js']}>
<Tab value="SDK (any)">
```ts
import { createClient } from '@dyrected/sdk'

const client = createClient({
  baseUrl: process.env.NEXT_PUBLIC_DYRECTED_URL!,
  apiKey: process.env.DYRECTED_API_KEY!,
})

const form = new FormData()
form.append('file', file)
form.append('alt', 'My awesome image')

const media = await client.collection('media').upload(form)
```
</Tab>
<Tab value="Next.js">
```ts
// Using raw fetch in a Client Component or Server Action
const form = new FormData()
form.append('file', file)
form.append('alt', file.name)

const res = await fetch(`${process.env.NEXT_PUBLIC_DYRECTED_URL}/collections/media`, {
  method: 'POST',
  headers: { 'x-api-key': process.env.DYRECTED_API_KEY! },
  body: form,
})
const media = await res.json()
```
</Tab>

## 5. Attach media to another collection

```ts
// dyrected.config.ts  (same for both frameworks)
{
  slug: 'posts',
  fields: [
    { name: 'title',      type: 'text' },
    { name: 'coverImage', type: 'relationship', relationTo: 'media' },
  ],
}
```

Fetch with `depth: 1` to inline the media document:

```ts
// docs[0].coverImage.url        → full storage URL
// docs[0].coverImage.sizes.thumbnail.url
```

## 6. Render images

<Tabs items={['Next.js', 'Nuxt.js']}>
<Tab value="Next.js">
```tsx
import Image from 'next/image'

export function PostCard({ post }: { post: any }) {
  const cover = post.coverImage
  if (!cover) return null

  return (
    <Image
      src={cover.sizes?.card?.url ?? cover.url}
      alt={cover.alt ?? ''}
      width={800}
      height={450}
    />
  )
}
```

> Add external storage domains to `images.remotePatterns` in `next.config.ts`.
</Tab>
<Tab value="Nuxt.js">
```vue
<!-- components/PostCard.vue -->
<script setup lang="ts">
defineProps<{ post: any }>()
</script>

<template>
  <img
    v-if="post.coverImage"
    :src="post.coverImage.sizes?.card?.url ?? post.coverImage.url"
    :alt="post.coverImage.alt ?? ''"
    width="800"
    height="450"
  />
</template>
```

> For Nuxt.js Image optimisation, use `<NuxtImg>` from `@nuxt/image` and add your storage domain to `image.domains` in `nuxt.config.ts`.
</Tab>
</Tabs>
